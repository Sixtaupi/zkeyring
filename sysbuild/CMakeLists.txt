function(${SYSBUILD_CURRENT_MODULE_NAME}_pre_cmake)
  if(SB_CONFIG_BOOTLOADER_MCUBOOT AND NOT SB_CONFIG_ZKEYRING_SIGNATURE_INHERIT)
    foreach(image mcuboot ${DEFAULT_IMAGE})
      get_property(config_scripts TARGET ${image} PROPERTY IMAGE_CONF_SCRIPT)
      list(PREPEND config_scripts "${CMAKE_CURRENT_FUNCTION_LIST_DIR}/signature.cmake")
      set_property(TARGET ${image} PROPERTY IMAGE_CONF_SCRIPT "${config_scripts}")
    endforeach()
  endif()
  sysbuild_cache_set(VAR ${DEFAULT_IMAGE}_ZKEYRING_USE_PROFILE_KEY
                     "${SB_CONFIG_ZKEYRING_PROFILE_PUBLIC_KEY}")
  if(NOT SB_CONFIG_ZKEYRING_PROFILE_PUBLIC_KEY)
    return()
  endif()

  # MCUboot configures first. Its build uses the PEM generated when the
  # application configures, after Zephyr has resolved the application's profile.
  ExternalProject_Get_Property(${DEFAULT_IMAGE} BINARY_DIR)
  set(key_files "${BINARY_DIR}/zkeyring-public.pem")

  # Keep any additional verification keys configured in Sysbuild.
  if(COMMAND sysbuild_mcuboot_resolve_signature_key_files)
    sysbuild_mcuboot_resolve_signature_key_files(configured_keys "${SB_CONFIG_BOOT_SIGNATURE_KEY_FILE}")
    string(REGEX MATCH ",.*$" extra_keys "${configured_keys}")
    string(APPEND key_files "${extra_keys}")
  endif()
  get_property(config TARGET mcuboot PROPERTY CONFIG)
  string(REGEX REPLACE "(^|\n)CONFIG_BOOT_SIGNATURE_KEY_FILE=[^\n]*\n" "\\1" config "${config}")
  set_property(TARGET mcuboot PROPERTY CONFIG "${config}")
  set_config_string(mcuboot CONFIG_BOOT_SIGNATURE_KEY_FILE "${key_files}")
endfunction()

function(${SYSBUILD_CURRENT_MODULE_NAME}_post_image_cmake)
  cmake_parse_arguments(POST "" "IMAGE" "IMAGES" ${ARGN})
  if(SB_CONFIG_BOOTLOADER_MCUBOOT AND POST_IMAGE STREQUAL "mcuboot")
    sysbuild_get(boot_pure IMAGE mcuboot VAR CONFIG_BOOT_SIGNATURE_TYPE_PURE KCONFIG)
    sysbuild_get(boot_sha384 IMAGE mcuboot VAR CONFIG_BOOT_IMG_HASH_ALG_SHA384 KCONFIG)
    sysbuild_get(boot_sha512 IMAGE mcuboot VAR CONFIG_BOOT_IMG_HASH_ALG_SHA512 KCONFIG)
    sysbuild_get(boot_rsa IMAGE mcuboot VAR CONFIG_BOOT_SIGNATURE_TYPE_RSA KCONFIG)
    sysbuild_get(boot_ed25519 IMAGE mcuboot VAR CONFIG_BOOT_SIGNATURE_TYPE_ED25519 KCONFIG)
    sysbuild_get(boot_p256 IMAGE mcuboot VAR CONFIG_BOOT_SIGNATURE_TYPE_ECDSA_P256 KCONFIG)
    if(boot_pure OR boot_sha384 OR boot_sha512)
      set(signature_type unsupported)
    elseif(boot_rsa)
      sysbuild_get(rsa_size IMAGE mcuboot VAR CONFIG_BOOT_SIGNATURE_TYPE_RSA_LEN KCONFIG)
      set(signature_type "rsa-${rsa_size}")
    elseif(boot_ed25519)
      set(signature_type ed25519)
    elseif(boot_p256)
      set(signature_type ecdsa-p256)
    else()
      set(signature_type none)
    endif()
    sysbuild_cache_set(VAR ${DEFAULT_IMAGE}_ZKEYRING_BOOT_SIGNATURE "${signature_type}")
  endif()
  if(SB_CONFIG_ZKEYRING_PROFILE_PUBLIC_KEY AND POST_IMAGE STREQUAL DEFAULT_IMAGE)
    # Configure both images before either builds when the profile or PEM changes.
    sysbuild_get(dependencies IMAGE ${DEFAULT_IMAGE} VAR ZKEYRING_PROFILE_KEY_DEPENDS CACHE)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${dependencies})
  endif()
endfunction()
